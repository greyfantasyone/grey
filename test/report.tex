\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}

\begin{document}
\begin{tikzpicture}[>=stealth, thick]

% ------------------- 定义样式 -------------------
% 左侧和右侧的普通方框 (加宽了一些)
\tikzstyle{box} = [draw, solid, thick, rectangle, align=center, font=\songti, inner sep=8pt, minimum height=1.2cm, minimum width=3.5cm]
% 中间的核心决策大方框 (加宽加高)
\tikzstyle{centerbox} = [draw, solid, thick, rectangle, align=left, font=\songti, inner sep=15pt, text width=7.5cm]
% 底部的规则文本框 (加宽)
\tikzstyle{textbox} = [draw, solid, thick, rectangle, align=left, font=\songti, inner sep=12pt, text width=14.5cm]

% ------------------- 坐标规划 -------------------
\def\xL{0}    % 左侧列 X 坐标
\def\xC{7.5}  % 中间列 X 坐标 (大幅拉开距离)
\def\xR{15}   % 右侧列 X 坐标 (大幅拉开距离)

% ------------------- 放置节点 -------------------

% 1. 左侧输入节点 (增加了 Y 轴的上下间距)
\node[box] (in1) at (\xL, 3) {目的主机 FSM 状态 \\ $S_d \in \{S_0, S_1, S_2, S_3\}$};
\node[box] (in2) at (\xL, 0) {业务类型标签 BT \\ (BE / DS / LC)};
\node[box] (in3) at (\xL, -3) {租户权重 $W(T_i)$ \\ (多租户策略参数)};

% 2. 中间处理节点
\node[centerbox] (center) at (\xC, 0) {
    \begin{center}\textbf{优先级映射与丢包决策中心}\end{center}\vspace{3mm}
    \textbf{1. 优先级降级计算：} \\[1mm]
    $\pi = \max(0, P_{base}(BT) - \lambda \cdot S_d)$ \\[4mm]
    \textbf{2. 丢包资格判定：} \\[1mm]
    $\delta = f(S_d, BT)$ \\[4mm]
    \textbf{3. 优先级冻结检查：} \\[1mm]
    IF $S_d = S_3$ THEN $\pi = \pi_f$
};

% 3. 右侧输出节点
\node[box] (out1) at (\xR, 1.5) {调度队列优先级 $\pi$ \\ (映射至硬件队列)};
\node[box] (out2) at (\xR, -1.5) {丢包资格标志位 $\delta$ \\ (0: 不可丢 / 1: 可丢)};

% 4. 底部文本说明节点 (大幅度下移到 y = -7.5，避免重叠)
\node[textbox] (rules) at (\xC, -7.5) {
    \textbf{调度决策执行规则：}\\[2mm]
    \textbf{规则一：}在重度拥塞（$S_d = S_2$）状态下，对于后台业务（BE），将其优先级降低至预设的低优先级级别并允许丢弃；对于延迟敏感业务（DS），降级其优先级但限制丢弃资格。\\[2mm]
    \textbf{规则二：}对于无损关键业务（LC），无论当前目的主机的拥塞状态如何，强制配置为不可丢包状态以禁止设备丢包。\\[2mm]
    \textbf{规则三：}当状态机进入恢复状态（$S_d = S_3$）时，系统启动优先级冻结机制，强制映射该流的优先级至历史冻结优先级 $\pi_f$，以维持数据包的先进先出（FIFO）语义。
};

% ------------------- 绘制连线 -------------------

% 左侧输入到中间 (箭头保持水平)
\draw[->] (in1.east) -- (in1.east -| center.west);
\draw[->] (in2.east) -- (center.west);
\draw[->] (in3.east) -- (in3.east -| center.west);

% 中间到右侧输出
\draw[->] ([yshift=1.5cm]center.east) -- (out1.west);
\draw[->] ([yshift=-1.5cm]center.east) -- (out2.west);

\end{tikzpicture}
