\begin{figure}[htbp]
\centering
\begin{tikzpicture}[auto, node distance=2cm, >=latex']
    % --- 定义图形样式 ---
    \tikzset{
        block/.style = {draw, rectangle, minimum height=3em, minimum width=4em, align=center},
        sum/.style = {draw, circle, inner sep=0mm, minimum size=6mm},
        branch/.style = {fill, shape=circle, minimum size=3pt, inner sep=0pt}
    }

    % --- 位置控制分支 (上侧通道, X/Y轴) ---
    \node [coordinate] (inputX) {};
    \node [right=0.5cm of inputX] (labelXd) {$X_d$};
    \node [sum, right=1.5cm of labelXd] (sumP) {};
    \node [block, right=1.2cm of sumP] (cp) {位置控制器\\$C_p$};
    \node [block, right=1.2cm of cp] (S) {选择矩阵\\$S$};

    % --- 力控制分支 (下侧通道, Z轴恒力) ---
    \node [coordinate, below=3.5cm of inputX] (inputF) {};
    \node [right=0.5cm of inputF] (labelFd) {$F_d$};
    \node [sum, right=1.5cm of labelFd] (sumF) {};
    \node [block, right=1.2cm of sumF] (cf) {力控制器\\$C_f$};
    \node [block, right=1.2cm of cf] (ImS) {选择矩阵\\$I-S$};

    % --- 综合指令与机器人本体 ---
    % 加和节点放在 S 和 I-S 中间的偏右位置
    \node [sum, right=1.5cm of S, yshift=-1.75cm] (sumTotal) {};
    \node [block, right=1.5cm of sumTotal] (robot) {笛卡尔机械臂\\(运动学/动力学)};

    % --- 环境模型与输出 ---
    \node [block, right=2cm of robot, yshift=-1.75cm] (env) {环境刚度\\$K_e$};
    \node [coordinate, right=4cm of robot] (outputX) {};

    % --- 连线：位置环 (前向) ---
    \draw [->] (labelXd) -- node[pos=0.8] {$+$} (sumP);
    \draw [->] (sumP) -- node {$E_p$} (cp);
    \draw [->] (cp) -- (S);
    \draw [->] (S) -| node[pos=0.9, left] {$+$} (sumTotal);

    % --- 连线：力控制环 (前向) ---
    \draw [->] (labelFd) -- node[pos=0.8] {$+$} (sumF);
    \draw [->] (sumF) -- node {$E_f$} (cf);
    \draw [->] (cf) -- (ImS);
    \draw [->] (ImS) -| node[pos=0.9, left] {$+$} (sumTotal);

    % --- 连线：控制输入与本体 ---
    \draw [->] (sumTotal) -- node {$\tau$ (驱动力)} (robot);

    % --- 连线：输出与位置反馈回路 ---
    \node [branch, right=1.5cm of robot] (branchX) {};
    \draw [-] (robot) -- node [above] {$X$} (branchX);
    \draw [->] (branchX) -- node [above] {$X$ (实际位置)} (outputX);
    
    % 从分支点向上拉出位置反馈线
    \node [coordinate, above=2cm of robot] (feedbackP) {};
    \draw [-] (branchX) |- (feedbackP);
    \draw [->] (feedbackP) -| node[pos=0.95, right] {$-$} (sumP);

    % --- 连线：环境接触力与力反馈回路 ---
    \draw [->] (branchX) |- (env); % 位置输入到环境产生力
    \node [branch, right=1cm of env] (branchF) {};
    \draw [->] (env) -- node [above] {$F$} (branchF) -- +(2,0) node[right] {$F$ (实际接触力)};
    
    % 从分支点向下拉出力反馈线
    \node [coordinate, below=1.5cm of env] (feedbackF) {};
    \draw [-] (branchF) |- (feedbackF);
    \draw [->] (feedbackF) -| node[pos=0.95, right] {$-$} (sumF);

\end{tikzpicture}
\caption{超声波检测平台力/位混合控制系统框图（笛卡尔空间）}
\end{figure}
